cmake_minimum_required(VERSION 3.16)
project(tidesense_dds_benchmark LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 14)
find_package(ament_cmake REQUIRED)
find_package(rcl REQUIRED)
find_package(rclc REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
add_executable(native_bench bench.cpp native_bridge.cpp)
target_compile_definitions(native_bench PRIVATE BENCH_NATIVE=1)
target_compile_options(native_bench PRIVATE -Wall -Wextra -Werror -O3)
ament_target_dependencies(native_bench rcl rclc geometry_msgs std_msgs)
target_link_libraries(native_bench pthread)

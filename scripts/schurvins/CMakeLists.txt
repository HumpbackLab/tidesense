set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
get_filename_component(SDK_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(SCHUR_SOURCE "${SDK_ROOT}/SchurVINS" CACHE PATH "Official SchurVINS checkout")
set(WORK "${SDK_ROOT}/install/schurvins-gcc11-static" CACHE PATH "Build dependencies/staging directory")
set(DEPS "${SDK_ROOT}/install/schurvins-build/deps" CACHE PATH "Dependency source directory")
option(SCHUR_OPT_HALFSAMPLE_NEON "Exact AArch64 NEON pyramid downsampling" OFF)

option(SCHUR_OPT_FAST10_NEON "Exact AArch64 NEON FAST-10 detection and scoring" OFF)
option(SCHUR_OPT_ALIGN2D_NEON "AArch64 NEON float patch alignment with ordered accumulation" OFF)
option(SCHUR_OPT_ALIGN2D_HESSIAN_NEON "Exact integer NEON align2D Hessian preparation" OFF)
option(SCHUR_OPT_WARP_AFFINE_NEON "Float NEON 10x10 affine patch warp" OFF)
option(SCHUR_OPT_WARP_TAIL2 "Two-lane NEON affine warp row tails" OFF)
option(SCHUR_SYNC_OUTLIER_OBSERVATIONS "Remove rejected frontend observations from the Schur backend" OFF)
option(SCHUR_BUILD_BENCHMARKS "Build optimization correctness/benchmark tool" OFF)
option(SCHUR_STATIC "Build a fully static executable" ON)
option(SCHUR_WITH_MICRO_ROS "Build optional micro-ROS telemetry (runtime default off)" OFF)
set(SCHUR_MICRO_ROS_ROOT "${SDK_ROOT}/install/micro-ros/firmware/build" CACHE PATH "A53 micro-ROS static library and headers")
set(GLOG_LIBRARY "${WORK}/stage/lib/libglog.a" CACHE FILEPATH "Target glog static library")
if(SCHUR_STATIC)
  set(CMAKE_SKIP_RPATH TRUE)
else()
  set(CMAKE_INSTALL_RPATH "$ORIGIN/../lib;$ORIGIN")
endif()
find_package(OpenCV 3.4 REQUIRED COMPONENTS core imgproc imgcodecs calib3d video highgui)
find_package(yaml-cpp REQUIRED)
set(MODULES rpg_common vikit/vikit_common vikit/vikit_cameras vikit/vikit_solver
    svo_common svo_vio_common svo_direct svo_tracker svo_img_align svo)
set(SOURCES)
foreach(module IN LISTS MODULES)
  include_directories("${SCHUR_SOURCE}/${module}/include")
  file(GLOB module_sources "${SCHUR_SOURCE}/${module}/src/*.cpp")
  list(APPEND SOURCES ${module_sources})
endforeach()
# These optional files require CUDA; the CPU image alignment implementation is built.
list(FILTER SOURCES EXCLUDE REGEX "/(frame_gpu|sparse_img_align_gpu|test_utils)\\.cpp$")
include_directories(${OpenCV_INCLUDE_DIRS} "${WORK}/sysroot/eigen/include/eigen3"
  "${WORK}/stage/include"
  "${DEPS}/minkindr/minkindr/include" "${DEPS}/fast_neon/include")
add_library(fast STATIC
  "${DEPS}/fast_neon/src/fast_9.cpp"
  "${DEPS}/fast_neon/src/fast_10.cpp"
  "${DEPS}/fast_neon/src/fast_10_score.cpp"
  "${DEPS}/fast_neon/src/nonmax_3x3.cpp")
# Keep the FAST-10 scalar reference for the default path and fallback.
# SCHUR_OPT_FAST10_NEON selects the matching NEON detector/scorer below.
# Eigen/OpenCV still use AArch64 NEON. ROS, loop closure, global map and
# mono OpenGV initialization are optional and are not enabled for this stereo build.
if(SCHUR_STATIC)
  add_library(schurvins STATIC ${SOURCES})
else()
  add_library(schurvins SHARED ${SOURCES})
endif()
target_compile_definitions(schurvins PUBLIC SVO_DEPTHFILTER_IN_REPROJECTOR)
if(SCHUR_SYNC_OUTLIER_OBSERVATIONS)
  set_property(SOURCE "${SCHUR_SOURCE}/svo/src/schur_vins.cpp"
    APPEND PROPERTY COMPILE_DEFINITIONS SCHUR_SYNC_OUTLIER_OBSERVATIONS=1)
endif()
foreach(opt HALFSAMPLE ZMSSD FAST10 ALIGN2D ALIGN2D_HESSIAN WARP_AFFINE)
  if(SCHUR_OPT_${opt}_NEON)
    if(NOT CMAKE_SYSTEM_PROCESSOR STREQUAL "aarch64")
      message(FATAL_ERROR "SCHUR_OPT_${opt}_NEON requires AArch64")
    endif()
    if(opt STREQUAL "ZMSSD")
      # Header implementation: propagate to all consumers to avoid ODR mismatch.
      target_compile_definitions(schurvins PUBLIC SCHUR_OPT_${opt}_NEON=1)
    elseif(opt STREQUAL "HALFSAMPLE")
      set_property(SOURCE "${SCHUR_SOURCE}/vikit/vikit_common/src/vision.cpp"
        APPEND PROPERTY COMPILE_DEFINITIONS SCHUR_OPT_HALFSAMPLE_NEON=1)
    elseif(opt STREQUAL "ALIGN2D" OR opt STREQUAL "ALIGN2D_HESSIAN")
      set_property(SOURCE "${SCHUR_SOURCE}/svo_direct/src/feature_alignment.cpp"
        APPEND PROPERTY COMPILE_DEFINITIONS SCHUR_OPT_${opt}_NEON=1)
    elseif(opt STREQUAL "WARP_AFFINE")
      set_property(SOURCE "${SCHUR_SOURCE}/svo_direct/src/patch_warp.cpp"
        APPEND PROPERTY COMPILE_DEFINITIONS SCHUR_OPT_WARP_AFFINE_NEON=1)
    else()
      set_property(SOURCE "${SCHUR_SOURCE}/svo_direct/src/feature_detection_utils.cpp"
        "${SCHUR_SOURCE}/svo_direct/src/fast10_neon.cpp"
        APPEND PROPERTY COMPILE_DEFINITIONS SCHUR_OPT_FAST10_NEON=1)
    endif()
  endif()
endforeach()
if(SCHUR_OPT_WARP_TAIL2)
  if(NOT SCHUR_OPT_WARP_AFFINE_NEON)
    message(FATAL_ERROR "SCHUR_OPT_WARP_TAIL2 requires SCHUR_OPT_WARP_AFFINE_NEON")
  endif()
  set_property(SOURCE "${SCHUR_SOURCE}/svo_direct/src/patch_warp.cpp"
    APPEND PROPERTY COMPILE_DEFINITIONS SCHUR_OPT_WARP_TAIL2=1)
endif()
if(SCHUR_BUILD_BENCHMARKS)
  add_executable(schurvins_warp_bench bench/warp.cpp bench/warp_reference.cpp)
  target_compile_definitions(schurvins_warp_bench PRIVATE

  add_executable(schurvins_align2d_bench bench/align2d.cpp bench/align2d_reference.cpp bench/align2d_hessian_only.cpp)
  target_compile_definitions(schurvins_align2d_bench PRIVATE
    SCHUR_ALIGNMENT_SOURCE="${SCHUR_SOURCE}/svo_direct/src/feature_alignment.cpp")
  target_link_libraries(schurvins_align2d_bench PRIVATE schurvins)
  if(SCHUR_STATIC)
    target_link_options(schurvins_align2d_bench PRIVATE -static -Wl,--gc-sections)
  endif()
  add_executable(schurvins_kernel_bench bench/kernels.cpp)
  target_link_libraries(schurvins_kernel_bench PRIVATE schurvins)
  foreach(opt HALFSAMPLE ZMSSD FAST10 ALIGN2D ALIGN2D_HESSIAN WARP_AFFINE)
    if(SCHUR_OPT_${opt}_NEON)
      target_compile_definitions(schurvins_kernel_bench PRIVATE SCHUR_OPT_${opt}_NEON=1)
    endif()
  endforeach()
  if(SCHUR_STATIC)
    target_link_options(schurvins_kernel_bench PRIVATE -static -Wl,--gc-sections)
  endif()
endif()

if(SCHUR_WITH_MICRO_ROS)
  if(NOT EXISTS "${SCHUR_MICRO_ROS_ROOT}/libmicroros.a")
    message(FATAL_ERROR "Build micro-ROS first: scripts/schurvins/micro_ros/build.sh")
  endif()
  target_compile_definitions(schurvins_micro_ros PRIVATE SCHUR_WITH_MICRO_ROS=1)
  target_include_directories(schurvins_micro_ros PRIVATE "${SCHUR_MICRO_ROS_ROOT}/include")
  # Humble installs some headers below an additional package directory.
  file(GLOB micro_ros_include_dirs LIST_DIRECTORIES TRUE "${SCHUR_MICRO_ROS_ROOT}/include/*")
  foreach(dir IN LISTS micro_ros_include_dirs)
    if(IS_DIRECTORY "${dir}")
      target_include_directories(schurvins_micro_ros PRIVATE "${dir}")
    endif()
  endforeach()
  target_link_libraries(schurvins_micro_ros PUBLIC "${SCHUR_MICRO_ROS_ROOT}/libmicroros.a" pthread rt m)
endif()

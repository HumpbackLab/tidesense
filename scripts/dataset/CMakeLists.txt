cmake_minimum_required(VERSION 3.13)
project(tidesense_dataset LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCV 3.4 REQUIRED COMPONENTS core)
find_package(yaml-cpp REQUIRED)
set(ZSTD_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/../../install/dataset-deps/zstd-1.5.7" CACHE PATH "Pinned zstd sources")
set(ZSTD_BUILD_PROGRAMS OFF CACHE BOOL "" FORCE)
set(ZSTD_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(ZSTD_BUILD_SHARED OFF CACHE BOOL "" FORCE)
set(ZSTD_LEGACY_SUPPORT OFF CACHE BOOL "" FORCE)
set(ZSTD_MULTITHREAD_SUPPORT OFF CACHE BOOL "" FORCE)
add_subdirectory("${ZSTD_SOURCE}/build/cmake" zstd EXCLUDE_FROM_ALL)
set(SENSOR "${CMAKE_CURRENT_SOURCE_DIR}/../schurvins")
set(TIDE_CAM_BUILD_EXAMPLES OFF)
add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../../tide_cam" tide_cam EXCLUDE_FROM_ALL)
add_executable(tidesense_record recorder.cpp ${SENSOR}/tide_camera.cpp ${SENSOR}/live_imu.cpp)
target_include_directories(tidesense_record PRIVATE ${SENSOR} ${OpenCV_INCLUDE_DIRS} "${ZSTD_SOURCE}/lib")
target_link_libraries(tidesense_record PRIVATE tide_cam::tide_cam ${OpenCV_LIBS} yaml-cpp libzstd_static pthread dl)
target_compile_options(tidesense_record PRIVATE -Wall -Wextra -Werror)
target_link_options(tidesense_record PRIVATE -static -Wl,--gc-sections)

option(DATASET_BUILD_TESTS "Build camera transport tests" OFF)
if(DATASET_BUILD_TESTS)
  add_executable(dataset_camera_mock ${SENSOR}/tests/camera_mock.c)
  target_include_directories(dataset_camera_mock PRIVATE ${SENSOR})
  target_link_options(dataset_camera_mock PRIVATE -static)
  add_executable(dataset_camera_test ${SENSOR}/tests/live_camera_test.cpp ${SENSOR}/live_camera.cpp)
  target_include_directories(dataset_camera_test PRIVATE ${SENSOR} ${OpenCV_INCLUDE_DIRS})
  target_link_libraries(dataset_camera_test PRIVATE ${OpenCV_LIBS} pthread dl)
  target_link_options(dataset_camera_test PRIVATE -static -Wl,--gc-sections)
endif()

cmake_minimum_required(VERSION 3.13)
project(tidesense_depth LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 14)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCV REQUIRED COMPONENTS core imgproc calib3d)
find_package(yaml-cpp REQUIRED)
set(SENSOR "${CMAKE_CURRENT_SOURCE_DIR}/../schurvins")
get_filename_component(SDK "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(all64_default OFF)
if(CMAKE_SYSTEM_PROCESSOR STREQUAL "aarch64")
  set(all64_default ON)
endif()
option(DEPTH_STEREOBM_ALL64 "Use audited A53 StereoBM optimizations in depth only" ${all64_default})
if(DEPTH_STEREOBM_ALL64)
  if(NOT CMAKE_SYSTEM_PROCESSOR STREQUAL "aarch64" OR NOT OpenCV_VERSION VERSION_EQUAL "3.4.20")
    message(FATAL_ERROR "all64 requires AArch64 OpenCV 3.4.20; use -DDEPTH_STEREOBM_ALL64=OFF otherwise")
  endif()
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  set(depth_bm_object "${CMAKE_CURRENT_BINARY_DIR}/stereobm-all64.o")
  add_custom_command(OUTPUT "${depth_bm_object}"
    COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/build-stereobm.py"
      --compiler "${CMAKE_CXX_COMPILER}" --output "${depth_bm_object}"
    DEPENDS build-stereobm.py
      "${SDK}/scripts/vio-compare/probes/stereo-bm/prepare-opt.py"
      "${SDK}/install/schurvins-build/deps/opencv-3.4.20/modules/calib3d/src/stereobm.cpp"
      "${SDK}/install/schurvins-gcc11-static/opencv/modules/calib3d/CMakeFiles/opencv_calib3d.dir/flags.make"
    VERBATIM)
  add_custom_target(depth_bm DEPENDS "${depth_bm_object}")
endif()
add_library(depth_engine STATIC depth.cpp)
target_include_directories(depth_engine PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} ${OpenCV_INCLUDE_DIRS})
if(DEPTH_STEREOBM_ALL64)
  add_dependencies(depth_engine depth_bm)
  # Resolve StereoBM symbols before the original calib3d archive is scanned.
  target_link_libraries(depth_engine PUBLIC "${depth_bm_object}")
endif()
target_link_libraries(depth_engine PUBLIC ${OpenCV_LIBS} yaml-cpp)
target_compile_options(depth_engine PRIVATE -Wall -Wextra -Werror)

option(DEPTH_BUILD_APP "Build the A53 micro-ROS depth application" ON)
option(DEPTH_STATIC "Link the A53 application statically" ON)
if(DEPTH_BUILD_APP)
  set(MICRO_ROS_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../install/micro-ros/firmware/build" CACHE PATH "A53 micro-ROS")
  if(NOT EXISTS "${MICRO_ROS_ROOT}/libmicroros.a")
    message(FATAL_ERROR "Build scripts/schurvins/micro_ros/build.sh first")
  endif()
  add_subdirectory("${SDK}/tide_cam" tide_cam EXCLUDE_FROM_ALL)
  add_executable(tidesense_depth main.cpp depth_ros.cpp ${SENSOR}/tide_camera.cpp)
  target_include_directories(tidesense_depth PRIVATE ${SENSOR} "${MICRO_ROS_ROOT}/include")
  file(GLOB ros_include_dirs LIST_DIRECTORIES TRUE "${MICRO_ROS_ROOT}/include/*")
  foreach(dir IN LISTS ros_include_dirs)
    if(IS_DIRECTORY "${dir}")
      target_include_directories(tidesense_depth PRIVATE "${dir}")
    endif()
  endforeach()
  target_compile_options(tidesense_depth PRIVATE -Wall -Wextra -Werror)
  target_link_libraries(tidesense_depth PRIVATE depth_engine tide_cam::tide_cam
    "${MICRO_ROS_ROOT}/libmicroros.a" pthread rt m dl)
  if(DEPTH_STATIC)
    target_link_options(tidesense_depth PRIVATE -static -Wl,--gc-sections)
  endif()
endif()

option(DEPTH_BUILD_TESTS "Build geometry and metric-depth checks" OFF)
if(DEPTH_BUILD_TESTS)
  enable_testing()
  add_executable(depth_test test_depth.cpp)
  target_link_libraries(depth_test PRIVATE depth_engine)
  target_compile_options(depth_test PRIVATE -Wall -Wextra -Werror)
  if(DEPTH_STATIC AND CMAKE_CROSSCOMPILING)
    target_link_options(depth_test PRIVATE -static -Wl,--gc-sections)
  endif()
  add_test(NAME depth_geometry COMMAND depth_test "${SENSOR}/tidesense.yaml")
  add_executable(verify_matcher verify_matcher.cpp)
  target_link_libraries(verify_matcher PRIVATE depth_engine)
  target_compile_options(verify_matcher PRIVATE -Wall -Wextra -Werror)
  if(DEPTH_STATIC AND CMAKE_CROSSCOMPILING)
    target_link_options(verify_matcher PRIVATE -static -Wl,--gc-sections)
  endif()
  if(DEPTH_BUILD_APP)
    add_executable(depth_ros_test test_depth_ros.cpp depth_ros.cpp)
    target_include_directories(depth_ros_test PRIVATE ${SENSOR} "${MICRO_ROS_ROOT}/include")
    foreach(dir IN LISTS ros_include_dirs)
      if(IS_DIRECTORY "${dir}")
        target_include_directories(depth_ros_test PRIVATE "${dir}")
      endif()
    endforeach()
    target_compile_options(depth_ros_test PRIVATE -Wall -Wextra -Werror)
    target_link_libraries(depth_ros_test PRIVATE depth_engine "${MICRO_ROS_ROOT}/libmicroros.a" pthread rt m dl)
    if(DEPTH_STATIC)
      target_link_options(depth_ros_test PRIVATE -static -Wl,--gc-sections)
    endif()
  endif()
endif()

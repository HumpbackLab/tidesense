cmake_minimum_required(VERSION 3.13)
project(tide_cam LANGUAGES C)
include(GNUInstallDirs)
find_package(Threads REQUIRED)
set(TIDE_CAM_BACKEND "SG2002" CACHE STRING "Hardware backend: SG2002 or TEST (host tests only)")
set_property(CACHE TIDE_CAM_BACKEND PROPERTY STRINGS SG2002 TEST)
option(TIDE_CAM_BUILD_EXAMPLES "Build capture example" ON)
option(TIDE_CAM_BUILD_TESTS "Build host contract tests" OFF)
option(TIDE_CAM_BUILD_BOARD_TESTS "Build real SG2002 lifecycle test" OFF)
get_filename_component(_sdk_default "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
set(TIDE_CAM_SDK_ROOT "${_sdk_default}" CACHE PATH "SG2002 SDK root")

add_library(tide_cam STATIC src/tide_cam.c src/config.c)
add_library(tide_cam::tide_cam ALIAS tide_cam)
target_compile_features(tide_cam PUBLIC c_std_11)
target_include_directories(tide_cam PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>
    $<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>
    PRIVATE src)
target_link_libraries(tide_cam PUBLIC Threads::Threads)
target_compile_options(tide_cam PRIVATE -Wall -Wextra -Werror)

if(TIDE_CAM_BACKEND STREQUAL "SG2002")
    if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(aarch64|arm64)$")
        message(FATAL_ERROR "SG2002 requires an AArch64 toolchain; use cmake/sg2002-gcc6.cmake")
    endif()
    set(_mpi "${TIDE_CAM_SDK_ROOT}/cvi_mpi")
    target_sources(tide_cam PRIVATE src/sg2002/backend.c)
    target_include_directories(tide_cam SYSTEM PRIVATE
        "${_mpi}/include" "${_mpi}/modules/isp/include/cv181x"
        "${_mpi}/sample/common" "${_mpi}/component/isp/common" "${_mpi}/3rdparty/inih")
    set(_sdk_libs sample sys vi vo vpss rgn gdc venc cvi_bin cvi_bin_isp isp isp_algo ae af awb sns_full ini)
    set(_archives)
    foreach(_name IN LISTS _sdk_libs)
        find_library(TIDE_CAM_SDK_${_name} NAMES "lib${_name}.a"
            PATHS "${_mpi}/lib" "${_mpi}/lib/3rd" NO_DEFAULT_PATH NO_CMAKE_FIND_ROOT_PATH)
        if(NOT TIDE_CAM_SDK_${_name})
            message(FATAL_ERROR "Missing SDK archive lib${_name}.a; run tide_cam/build.sh first")
        endif()
        list(APPEND _archives "${TIDE_CAM_SDK_${_name}}")
    endforeach()
    target_link_libraries(tide_cam PUBLIC -Wl,--start-group ${_archives} -Wl,--end-group m dl atomic rt)
    target_link_options(tide_cam INTERFACE -Wl,--wrap=CVI_VI_SetChnAttr)
elseif(TIDE_CAM_BACKEND STREQUAL "TEST" AND TIDE_CAM_BUILD_TESTS)
    target_sources(tide_cam PRIVATE tests/backend_mock.c)
    target_include_directories(tide_cam PRIVATE tests)
else()
    message(FATAL_ERROR "Unsupported backend (TEST is only available with TIDE_CAM_BUILD_TESTS=ON)")
endif()

if(TIDE_CAM_BUILD_EXAMPLES)
    add_executable(tide_cam_capture examples/capture.c)
    target_link_libraries(tide_cam_capture PRIVATE tide_cam::tide_cam)
    target_compile_options(tide_cam_capture PRIVATE -Wall -Wextra -Werror)
endif()
if(TIDE_CAM_BUILD_TESTS)
    if(NOT TIDE_CAM_BACKEND STREQUAL "TEST")
        message(FATAL_ERROR "Contract tests must use the TEST backend")
    endif()
    enable_testing()
    add_executable(tide_cam_test tests/test_capture.c)
    target_include_directories(tide_cam_test PRIVATE tests)
    target_link_libraries(tide_cam_test PRIVATE tide_cam)
    target_compile_options(tide_cam_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    add_test(NAME tide_cam_contract COMMAND tide_cam_test)
    set_tests_properties(tide_cam_contract PROPERTIES TIMEOUT 20)
endif()
if(TIDE_CAM_BUILD_BOARD_TESTS)
    if(NOT TIDE_CAM_BACKEND STREQUAL "SG2002")
        message(FATAL_ERROR "Board tests require SG2002")
    endif()
    add_executable(tide_cam_board_test tests/board_lifecycle.c)
    target_link_libraries(tide_cam_board_test PRIVATE tide_cam)
    target_compile_options(tide_cam_board_test PRIVATE -Wall -Wextra -Werror -UNDEBUG)
endif()
if(TIDE_CAM_BACKEND STREQUAL "SG2002")
    install(TARGETS tide_cam EXPORT tide_camTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
    install(FILES include/tide_cam.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
    install(EXPORT tide_camTargets NAMESPACE tide_cam:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/tide_cam)
    install(FILES cmake/tide_camConfig.cmake DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/tide_cam)
    if(TIDE_CAM_BUILD_EXAMPLES)
        install(TARGETS tide_cam_capture RUNTIME DESTINATION ${CMAKE_INSTALL_BINDIR})
    endif()
endif()
